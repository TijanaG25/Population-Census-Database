1.	Приказати укупан број економско активног становништва.
select sum(ukupno)
from ekonomskiAktivni

2.	Приказати колико Срба говори црногорски језик у Републици Србији.
select pomNPJezik.broj
from pomNPJezik join nacionalnePripadnosti on idNacionalnaPripadnost=nacionalnePripadnosti.id
				join jezici on idJezika=jezici.id
				join regioni on idRegiona=regioni.id
where dbo.fn_latinica(nacionalnePripadnosti.naziv)='Srbi'  and dbo.fn_latinica(jezici.naziv)='Crnogorski' and dbo.fn_latinica(regioni.naziv)='Republika Srbija'

3.	Колико има католичког становништва у Војводини?
 select max(broj)
from pomNPVeroispovest join veroispovesti on idVeroispovesti=veroispovesti.id
			        join regioni on idRegiona=regioni.id
where (dbo.fn_latinica(regioni.naziv) LIKE 'Region Vojvodine') and (dbo.fn_latinica(veroispovesti.naziv) LIKE 'Katolička');

4.	Које је најзаступљеније занимање у Београдском региону?
select top(1) sum(pomZanimanjeDelatnosti.broj),zanimanja.naziv
from pomZanimanjeDelatnosti join regioni on idRegiona=regioni.id
				         join zanimanja on idZanimanja=zanimanja.id
where dbo.fn_latinica(regioni.naziv) like 'Beogradski region' and (zanimanja.id!=13)
group by(zanimanja.naziv)
order by sum(pomZanimanjeDelatnosti.broj) desc

5.	Колико има административних службеника?
select  sum(pomZanimanjeDelatnosti.broj),zanimanja.naziv
from pomZanimanjeDelatnosti join zanimanja on idZanimanja=zanimanja.id
where dbo.fn_latinica(zanimanja.naziv) like 'Administrativni službenici'
group by(zanimanja.naziv)

6.	Колико становништва се бави пољопривредом?
select  sum(pomZanimanjeDelatnosti.broj),delatnosti.naziv
from pomZanimanjeDelatnosti join delatnosti on idDelatnosti=delatnosti.id
where dbo.fn_latinica(delatnosti.naziv) like 'Poljoprivreda, šumarstvo, ribarstvo '
group by(delatnosti.naziv)

7.	У ком региону је највише незапосленог становништва?
select  top(1) sum(ekonomskiAktivni.ukupnoNezaposleni),regioni.naziv
from pomEkonomski join ekonomskiAktivni on idAktivni=ekonomskiAktivni.id
				join regioni on idRegiona=regioni.id
where dbo.fn_latinica(regioni.naziv) not like 'Republika Srbija'
group by(regioni.naziv)
order by sum(ekonomskiAktivni.ukupnoNezaposleni) desc

8.	Колико је пензионера било 2011. године?
select  sum(penzioneri)
from ekonomskiNeaktivni

9.	Приказати број становника од 30 до 34 године из Шумадије и Западне Србије, који су по занимању административни службеници? (id за године је 5)
select sum(pomZanimanjeDelatnosti.broj), zanimanja.naziv
from pomZanimanjeDelatnosti join regioni on idRegiona= regioni.id
			join pomPolGodine on regioni.id=pomPolGodine.idRegiona
			join godine on  pomPolGodine.idGodine=godine.id
			join zanimanja on idZanimanja=zanimanja.id
where dbo.fn_latinica(zanimanja.naziv) like 'Administrativni službenici' and godine.id=5 and dbo.fn_latinica(regioni.naziv) like 'Region Šumadije i Zapadne Srbije'
group by zanimanja.naziv

10.	Приказати занимање које је најзаступљеније међу становништвом од 25 до 29 година. (id за године је 4)
select top(1) sum(pomZanimanjeDelatnosti.broj), zanimanja.naziv
from pomZanimanjeDelatnosti join regioni on idRegiona= regioni.id
				join pomPolGodine on regioni.id=pomPolGodine.idRegiona
				join godine on  pomPolGodine.idGodine=godine.id
				join zanimanja on idZanimanja=zanimanja.id
where godine.id=4 and zanimanja.id!=13
group by zanimanja.naziv 
order by sum(pomZanimanjeDelatnosti.broj) desc

11.	Приказати језике које говори становништво исламске вероисповести у Војводини.
select jezici.naziv
from pomNPJezik join jezici on idJezika=jezici.id
				join regioni on idRegiona=regioni.id
		join pomNPVeroispovest on regioni.id=pomNPVeroispovest.idRegiona
	join veroispovesti on pomNPVeroispovest.idVeroispovesti=veroispovesti.id
where dbo.fn_latinica(regioni.naziv) like 'Region Vojvodine'and dbo.fn_latinica(veroispovesti.naziv) like 'Islamska'
group by (jezici.naziv)

12.	Која је најзаступљенија делатност у Источној Србији?
select top(1) sum(pomZanimanjeDelatnosti.broj),delatnosti.naziv
from pomZanimanjeDelatnosti join delatnosti on idDelatnosti=delatnosti.id
					join regioni on idRegiona=regioni.id 
where dbo.fn_latinica(regioni.naziv) like 'Region Južne i Istočne Srbije' and delatnosti.id!=1
group by delatnosti.naziv
order by sum(pomZanimanjeDelatnosti.broj) desc
